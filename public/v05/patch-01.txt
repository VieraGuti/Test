function v05Box(w,h,d,x,y,z,mat){const m=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),mat);m.position.set(x,y,z);m.castShadow=true;m.receiveShadow=true;return m}
function v05AddCardinalObstacle(x,z,w,d,rot,tag='build'){
  const q=Math.abs(Math.sin(rot));
  const ww=q>.7?d:w, dd=q>.7?w:d;
  return addObstacle(x-ww/2,x+ww/2,z-dd/2,z+dd/2,tag);
}
function v05AddFrameObstacles(x,z,rot){
  const sideW=.66, depth=.28;
  const local=[-1.17,1.17];
  const obs=[];
  for(const lx of local){
    const wx=x+Math.cos(rot)*lx, wz=z-Math.sin(rot)*lx;
    obs.push(v05AddCardinalObstacle(wx,wz,sideW,depth,rot,'build'));
  }
  return obs;
}
function v05StructureGroup(type,x,z,rot){
  const g=new THREE.Group();g.position.set(x,0,z);g.rotation.y=rot;
  const wood=v05Wood(), dark=v05DarkWood(), metal=v05Metal();
  if(type==='wall'){
    g.add(v05Box(3,2.18,.18,0,1.09,0,wood));
    for(const lx of [-1.42,0,1.42])g.add(v05Box(.10,2.34,.26,lx,1.17,0,dark));
    g.add(v05Box(3,.12,.27,0,.08,0,dark),v05Box(3,.12,.27,0,2.27,0,dark));
  }else if(type==='windowwall'){
    g.add(v05Box(.72,2.18,.18,-1.14,1.09,0,wood),v05Box(.72,2.18,.18,1.14,1.09,0,wood));
    g.add(v05Box(1.58,.55,.18,0,.275,0,wood),v05Box(1.58,.50,.18,0,2.10,0,wood));

  }else if(type==='doorframe'){
    g.add(v05Box(.76,2.18,.18,-1.12,1.09,0,wood),v05Box(.76,2.18,.18,1.12,1.09,0,wood));

spawnStructure=function(type,x,z,rot=0,id=`b-${Date.now()}`){
  if(!V05_BUILD_TYPES.includes(type))return v05SpawnStructureBase(type,x,z,rot,id);
  rot=Math.round(rot/(Math.PI/2))*(Math.PI/2);
  if(type==='door'){
    const width=1.35,height=2.08;
    const pivot=new THREE.Group();
    const hx=x-Math.cos(rot)*width/2, hz=z+Math.sin(rot)*width/2;
    pivot.position.set(hx,0,hz);pivot.rotation.y=rot;scene.add(pivot);
    const doorMat=v05Wood(),panel=v05Box(width,height,.10,width/2,height/2,0,doorMat);pivot.add(panel);
    panel.add(v05Box(.06,.06,.06,width*.34,0,.08,new THREE.MeshStandardMaterial({color:0xb49a65,metalness:.65,roughness:.35})));
    const obs=v05AddCardinalObstacle(x,z,width,.16,rot,'build-door');
    const d={type:'door',id:`built-${id}`,pivot,mesh:panel,obs,x,z,axis:Math.abs(Math.sin(rot))>.7?'z':'x',open:false,swing:1,baseRot:rot,v05:true};doors.push(d);
    const st={id,type,x,z,rot,group:pivot,obs,door:d,preview:false};structures.push(st);return st;
  }
  const g=v05StructureGroup(type,x,z,rot);scene.add(g);let obs=null;
  if(type==='doorframe')obs=v05AddFrameObstacles(x,z,rot);
  else if(type==='roof')obs=null;
  else obs=v05AddCardinalObstacle(x,z,BUILD_RECIPES[type].w,BUILD_RECIPES[type].d,rot,'build');
  const st={id,type,x,z,rot,group:g,obs,preview:false};structures.push(st);
  if(type==='stash')containers.push({type:'container',id:`stash-${id}`,label:'Cofre construido',x,z,group:g,lid:null,items:Object.assign({},containerSave[`stash-${id}`]||{})});
  return st;
};

function v05StaticBlocked(x,z,w,d,rot){
  const swap=Math.abs(Math.sin(rot))>.7,ww=swap?d:w,dd=swap?w:d;
  const minX=x-ww/2-.08,maxX=x+ww/2+.08,minZ=z-dd/2-.08,maxZ=z+dd/2+.08;
  for(const o of obstacles){
    if(!o.active||o.tag==='build'||o.tag==='build-door'||o.tag==='resource')continue;
    if(maxX>o.minX&&minX<o.maxX&&maxZ>o.minZ&&minZ<o.maxZ)return true;
  }
  return false;
}
function v05StructureTooClose(type,x,z){
  const limits={stash:1.25,campfire:1.2,barricade:.75,wall:1.0,windowwall:1.0,doorframe:1.0,roof:.7,door:.1};
  const lim=limits[type]??.8;
  return structures.some(s=>s.type!=='roof'&&Math.hypot(s.x-x,s.z-z)<lim);
}
