const V057_CYCLE_SECONDS=480;
const v057NightSky=new THREE.Color(0x101820),v057DaySky=new THREE.Color(0x9ba6a1),v057DawnSky=new THREE.Color(0x716d64),v057SkyTmp=new THREE.Color();

function v057Smooth(t){t=clamp(t,0,1);return t*t*(3-2*t)}
function v057ClockFromRuntime(){
  const t=((runTime%V057_CYCLE_SECONDS)+V057_CYCLE_SECONDS)%V057_CYCLE_SECONDS;
  if(t<45)return 5.5+(t/45)*1.5;                 // 05:30 -> 07:00 dawn
  if(t<315)return 7+((t-45)/270)*11.5;          // 07:00 -> 18:30 long day
  if(t<360)return 18.5+((t-315)/45)*1.5;        // 18:30 -> 20:00 dusk
  const h=20+((t-360)/120)*9.5;                 // 20:00 -> 05:30 short night
  return h>=24?h-24:h
}
function v057LightForHour(h){
  if(h>=7&&h<18.5)return 1;
  if(h>=5.5&&h<7)return .10+.90*v057Smooth((h-5.5)/1.5);
  if(h>=18.5&&h<20)return .10+.90*(1-v057Smooth((h-18.5)/1.5));
  return .10
}

updateWorldTime=function(){

};

function v057Wrap(a){while(a>Math.PI)a-=Math.PI*2;while(a<-Math.PI)a+=Math.PI*2;return a}

  const n=v057ObjectiveNav||$('v057-objective-nav');if(!n)return;
  const target=v057ObjectivePosition();if(!target){n.classList.remove('show');return}
  const src=inFlyHeli&&flyHeli?flyHeli.group.position:player.pos,heading=inFlyHeli&&flyHeli?flyHeli.heading:yaw;
  const dx=target.x-src.x,dz=target.z-src.z,d=Math.hypot(dx,dz),targetHeading=Math.atan2(-dx,-dz),rel=v057Wrap(targetHeading-heading);
  const arrow=n.querySelector('.v057-nav-arrow'),copy=n.querySelector('small');if(arrow)arrow.style.transform=`rotate(${rel}rad)`;

  n.classList.add('show');
}
