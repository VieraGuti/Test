function v058MatchChunkVisual(chunk,o){
  const x=(o.minX+o.maxX)/2,z=(o.minZ+o.maxZ)/2;let best=null,bestD=1e9;
  for(const child of chunk.group.children){
    if(!child||!child.position)continue;
    const d=Math.hypot(child.position.x-x,child.position.z-z);
    if(d<bestD){bestD=d;best=child}
  }
  return bestD<1.6?best:null
}
