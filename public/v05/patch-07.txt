function v053Wrap(a){while(a>Math.PI)a-=Math.PI*2;while(a<-Math.PI)a+=Math.PI*2;return a}
function v053PilotHeliNav(){
  const n=$('pilot-nav-v053');
  if(!n||!flyHeli||inFlyHeli){n?.classList.remove('show');return}
  const dx=flyHeli.group.position.x-player.pos.x,dz=flyHeli.group.position.z-player.pos.z,d=Math.hypot(dx,dz);
  const target=Math.atan2(-dx,-dz),rel=v053Wrap(target-yaw),arrow=n.querySelector('.v053-pilot-arrow'),small=n.querySelector('small');
  if(arrow)arrow.style.transform=`rotate(${rel*180/Math.PI}deg)`;
  if(small)small.textContent=d<4?'PULSA USAR PARA PILOTAR':`${Math.round(d)} m · HELI PILOTABLE`;
  n.classList.add('show');
}

function v053MapBounds(){
  const pts=[...v052Discovered].map(k=>k.split(',').map(Number));
  const pcx=v052ChunkCoord(player.pos.x),pcz=v052ChunkCoord(player.pos.z);if(!pts.length)pts.push([pcx,pcz]);
  let minX=Math.min(...pts.map(p=>p[0]),pcx)-1,maxX=Math.max(...pts.map(p=>p[0]),pcx)+1,minZ=Math.min(...pts.map(p=>p[1]),pcz)-1,maxZ=Math.max(...pts.map(p=>p[1]),pcz)+1;
  if(flyHeli){const hx=v052ChunkCoord(flyHeli.group.position.x),hz=v052ChunkCoord(flyHeli.group.position.z);minX=Math.min(minX,hx-1);maxX=Math.max(maxX,hx+1);minZ=Math.min(minZ,hz-1);maxZ=Math.max(maxZ,hz+1)}
  if(heliEvent?.active&&!heliEvent.looted){const hx=v052ChunkCoord(heliEvent.x),hz=v052ChunkCoord(heliEvent.z);minX=Math.min(minX,hx-1);maxX=Math.max(maxX,hx+1);minZ=Math.min(minZ,hz-1);maxZ=Math.max(maxZ,hz+1)}
  return {pts,minX,maxX,minZ,maxZ};
}
