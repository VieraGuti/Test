function v054Landed(h=flyHeli){return !!h&&h.group.position.y<=.92&&Math.abs(h.speed)<1.5&&Math.abs(h.verticalSpeed)<1.2}
function v054SetEngine(on){
  const h=flyHeli;if(!h)return;
  if(on){
    if(h.health<=0){toast('MOTOR BLOQUEADO · REPARA EL HELICÓPTERO','bad');return}
    if(h.fuel<=.5){toast('SIN COMBUSTIBLE','bad');return}
    if(h.engineState==='on'||h.engineState==='starting')return;
    h.engineState='starting';h.engineStartAt=now()+1.15;h.engine=false;AudioFX.burst(.12,.42,380,62);toast('ARRANCANDO TURBINA…','good');
  }else{
    h.engine=false;h.engineState='off';h.engineStartAt=0;AudioFX.click();toast('MOTOR APAGADO');
  }
}
function v054DamageHeli(h,amount,reason='IMPACTO'){

  if(h.health<=0){h.engine=false;h.engineState='off';toast('HELICÓPTERO AVERIADO · NECESITA REPARACIÓN','bad')}
}
