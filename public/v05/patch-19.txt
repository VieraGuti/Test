/* Fill the requested footprint first, then set a sane height. This avoids the old
   behaviour where a tall source model made the whole building tiny. */
function v0514FitFootprint(obj,w,d,h,{allowRotate=true}={}){
  if(!obj)return obj;obj.updateMatrixWorld(true);let b=new THREE.Box3().setFromObject(obj),s=b.getSize(new THREE.Vector3());
  if(!(s.x>.001&&s.y>.001&&s.z>.001))return obj;
  if(allowRotate&&w>0&&d>0){
    const tr=w/d,nr=s.x/s.z,sr=s.z/s.x,en=Math.abs(Math.log(Math.max(.001,nr/tr))),es=Math.abs(Math.log(Math.max(.001,sr/tr)));
    if(es+.08<en){obj.rotation.y+=Math.PI/2;obj.updateMatrixWorld(true);b=new THREE.Box3().setFromObject(obj);s=b.getSize(new THREE.Vector3())}
  }
  const k=Math.min(w/Math.max(.001,s.x),d/Math.max(.001,s.z));obj.scale.multiplyScalar(k);obj.updateMatrixWorld(true);
  b=new THREE.Box3().setFromObject(obj);s=b.getSize(new THREE.Vector3());if(h>0&&s.y>.001)obj.scale.y*=h/s.y;
  return v0514BottomAnchor(obj)
}
